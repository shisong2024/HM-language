-- -- predefined types

data Unit = Unit;

data Maybe a = None | Just a;
data Either a b = Left a | Right b;

data Ordering = LT | EQ | GT;

ability Ord a = Ord { cmp :: a -> a -> Ordering };
ability Show a = Show { show' :: a -> String };

def show {s :: Show a} = s.show';

-- -- functions of Unit

implicit def ordUnit = Ord { cmp = lambda x y -> EQ } :: Ord Unit;
implicit def showUnit = Show { show' = lambda x -> "Unit" } :: Show Unit;

-- -- functions of numbers

def mod q m = q - (q / m) * m;
def divmod m n = (m / n, mod m n);

def even i = mod i 2 == 0;
def odd i = mod i 2 != 0;

def abs n = if n < 0 then -n else n;
def signum i = if i < 0 then -1 else if i > 0 then 1 else 0;

def gcd m n = if n == 0 then abs m else gcd n (mod m n);

def _ordFromInt n = match n with -1 -> LT | 0 -> EQ | _ -> GT;
def _ord2Int o = match o with LT -> -1 | EQ -> 0 | GT -> 1;

implicit def ordInt = Ord { cmp = lambda x y -> _ordFromInt (intCompare x y) } :: Ord Int;

def _showInt n = let sign = intCompare n 0 in
    if sign == -1 then "-" ++ _showInt (0 - n) 
    else let res = intCompare n 10 in if res == -1 then fromCode (n + 48) 
    else _showInt (n / 10) ++ fromCode (n - (n / 10) * 10 + 48)
    ;

implicit def showInt = Show { show' = _showInt } :: Show Int;

-- -- functions of bool

def and x y = match (x, y) with (true, true) -> true | _ -> false;
def or x y = match (x, y) with (false, false) -> false | _ -> true;
def not x = match x with true -> false | false -> true;

implicit def ordBool = Ord { cmp = lambda a b -> match (a, b) with (true, false) -> GT | (false, true) -> LT | _ -> EQ } :: Ord Bool;
implicit def showBool = Show { show' = lambda x -> if x then "true" else "false" } :: Show Bool;

-- -- functions of list

def head l = match l with [] -> None | x: xs -> Just x;
def tail l = match l with [] -> [] | x: xs -> xs;
def last l = match l with [] -> None | x: [] -> Just x | x: xs -> last xs;

def null l = match l with [] -> true | _ -> false;

def map f l = match l with [] -> [] | x: xs -> f x: map f xs;
def filter f l = match l with [] -> [] | x: xs -> if f x then x: filter f xs else filter f xs;
def foldr f z l = match l with [] -> z | x: xs -> f x (foldr f z xs);
def foldl f z l = match l with [] -> z | x: xs -> foldl f (f z x) xs;
def sum l = foldr (+) 0 l;

def elem e l = match l with [] -> false | x: xs -> or (e == x) (elem e xs);

def any l = match l with [] -> false | x: xs -> if x then true else any xs;
def all l = match l with [] -> true | x: xs -> if not x then false else all xs;

def has f l = any (map f l);
def index n l = if n < 0 then None else match l with [] -> None | x: xs -> if n == 0 then Just x else index (n - 1) xs;
def findMaybe = let _find = lambda n a l -> match l with [] -> None | x: xs -> if x == a then Just n else _find (n + 1) a xs in _find 0;

def append fl bl = match fl with [] -> bl | x: xs -> x: append xs bl;
def snoc a l = match l with [] -> [a] | x: xs -> x: snoc a xs;

def zip f s = match (f, s) with ([], _) -> [] | (_, []) -> [] | (x: xs, y: ys) -> (x, y): zip xs ys;
def zipWith f fl bl = match (fl, bl) with ([], _) -> [] | (_, []) -> [] | (x: xs, y: ys) -> f x y: zipWith f xs ys;
def unzip l = match l with [] -> ([], []) | (x, y): rs -> let res = unzip rs in match res with (f, s) -> (x: f, y: s);

def range' a b = if a >= b then [] else a: range' (a + 1) b;
def range = range' 0;

def length l = match l with [] -> 0 | x: xs -> 1 + length xs;

def concat l = foldr append [] l;
def concatMap f l = foldr (lambda x acc -> append (f x) acc) [] l;

def take n l = if n <= 0 then [] else match l with [] -> [] | x: xs -> x: take (n - 1) xs;
def drop n l = if n <= 0 then l else match l with [] -> [] | x: xs -> drop (n - 1) xs;
def takeWhile f l = match l with [] -> [] | x: xs -> if f x then [] else x: takeWhile f xs;
def dropWhile f l = match l with [] -> [] | x: xs -> if f x then l else dropWhile f xs;

def replicate n x = if n <= 0 then [] else x: replicate (n - 1) x;

def nub l = match l with [] -> [] | x: xs -> x: nub (filter (!= x) xs);

def sort l = match l with 
      [] -> [] 
    | x: xs -> concat 
        [ sort (filter (<= x) xs)
        , [x]
        , sort (filter (> x) xs)
        ]
    ;

def _ordList {o :: Ord a} f b = match (f, b) with
      ([], []) -> EQ
    | (_, []) -> GT
    | ([], _) -> LT
    | (x: xs, y: ys) -> let res = o.cmp x y in
        match res with
              EQ -> _ordList f b
            | _ -> res
    ;

implicit def ordList = Ord { cmp = _ordList } :: Ord [a];

def _showList l = 
    let step = lambda x sl -> (match uncons sl with
          None -> show x
        | _    -> show x ++ "," ++ sl)
    in "[" ++ foldr step "" l ++ "]"
    ;

implicit def showList = Show { show' = _showList } :: Show [a];

-- -- combinators

def id a = a;
def flip f y x = f x y;
def compose g f x = g (f x);
def curry f x y = f (x, y);
def uncurry f tp = match tp with (x, y) -> f x y;

def (=>) x f = f x;
infixl 6 =>;


-- -- functions of tuples

def fst t = match t with (a, b) -> a;
def snd t = match t with (a, b) -> b;
def swap t = match t with (a, b) -> (b, a);

def _showTuple2 t = match t with
    (x, y) -> "(" ++ show x ++ "," ++ show y ++ ")"
    ;

implicit def showTuple2 = Show { show' = _showTuple2 } :: Show (a, b);

-- -- functions of datas

def isNone ma = match ma with None -> true | Just _ -> false;
def isJust ma = match ma with Just _ -> true | None -> false;

def fromMaybe ma d = match ma with None -> d | Just a -> a;
def fmapMaybe f ma = match ma with None -> None | Just a -> Just (f a);
def mapMaybe f l = match l with [] -> [] | x: xs -> match f x with None -> mapMaybe f xs | Just y -> y: mapMaybe f xs;

def _ordMaybe {o :: Ord a} mx my = match (mx, my) with
      (None, None) -> EQ
    | (_, None) -> GT
    | (None, _) -> LT
    | (Just x, Just y) -> o.cmp x y
    ;

implicit def ordMaybe = Ord { cmp = _ordMaybe } :: Ord (Maybe a);

def _showMaybe mx = match mx with
      None -> "None"
    | Just x -> "Just " ++ show x
    ;

implicit def showMaybe = Show { show' = _showMaybe } :: Show (Maybe a);

def isLeft e = match e with Left _ -> true | Right _ -> false;
def isRight e = match e with Left _ -> false | Right _ -> true;
def either f g e = match e with Left x -> f x | Right y -> g y;

def mapLeft f e = match e with Left x -> Left (f x) | _ -> e;
def mapRight f e = match e with Right x -> Right (f x) | _ -> e;

def _ordEither {ol :: Ord a} {or :: Ord b} ea eb = match (ea, eb) with
      (Left _, Right _) -> LT
    | (Right _, Left _) -> GT
    | (Left a, Left b) -> ol.cmp a b
    | (Right a, Right b) -> or.cmp a b
    ;

implicit def ordEither = Ord { cmp = _ordEither } :: Ord (Either a b);

def _showEither ex = match ex with
      Left x -> "Left " ++ show x
    | Right y -> "Right " ++ show y
    ;

implicit def showEither = Show { show' = _showEither } :: Show (Either a b);


-- -- functions of Ordering

implicit def ordOrdering = Ord { cmp = lambda x y -> _ordFromInt (intCompare (_ord2Int x) (_ord2Int y)) } :: Ord Ordering;
implicit def showOrdering = Show { show' = lambda x -> match x with LT -> "LT" | EQ -> "EQ" | GT -> "GT" } :: Show Ordering;

def min {o :: Ord a} x y = match o.cmp x y with LT -> x | _ -> y;
def max {o :: Ord a} x y = match o.cmp x y with GT -> x | _ -> y;

def minimum l = let _m = lambda res l -> match l with [] -> res | x: xs -> _m (min x res) xs in match l with [] -> None | x: xs -> Just (_m x xs);
def maximum l = let _m = lambda res l -> match l with [] -> res | x: xs -> _m (max x res) xs in match l with [] -> None | x: xs -> Just (_m x xs);

def (<) {o :: Ord a} x y = match o.cmp x y with LT -> true | _ -> false;
def (>) {o :: Ord a} x y = match o.cmp x y with GT -> true | _ -> false;
def (<=) {o :: Ord a} x y = match o.cmp x y with GT -> false | _ -> true;
def (>=) {o :: Ord a} x y = match o.cmp x y with LT -> false | _ -> true;


-- -- functions of String

def strLen s = match uncons s with None -> 0 | Just (_, rest) -> 1 + strLen rest;
def strConcat ss = match ss with [] -> "" | s: rest -> s ++ strConcat rest;
def strReverse s = match uncons s with None -> "" | Just (c, rest) -> strReverse rest ++ fromCode c;
def strTake n s = if n <= 0 then "" else (match uncons s with None -> "" | Just (c, rest) -> fromCode c ++ strTake (n - 1) rest);
def strDrop n s = if n <= 0 then s else (match uncons s with None -> "" | Just (_, rest) -> strDrop (n - 1) rest);
def strIndex n s = if n < 0 then None else (match uncons s with None -> None | Just (c, rest) -> if n == 0 then Just c else strIndex (n - 1) rest);
def strElem c s = match uncons s with None -> false | Just (d, rest) -> or (c == d) (strElem c rest);
def strMap f s = match uncons s with None -> "" | Just (c, rest) -> fromCode (f c) ++ strMap f rest;
def strFilter p s = match uncons s with None -> "" | Just (c, rest) -> if p c then fromCode c ++ strFilter p rest else strFilter p rest;

def strSplitAt n s = 
    if n <= 0 
        then ("", s) 
    else (match uncons s with 
          None -> ("", "") 
        | Just (c, rest) -> let (a, b) = strSplitAt (n - 1) rest in (fromCode c ++ a, b))
    ;

def isDigit c = and (c >= 48) (c <= 57);
def isUpper c = and (c >= 65) (c <= 90);
def isLower c = and (c >= 97) (c <= 122);
def isAlpha c = or (isUpper c) (isLower c);
def isSpace c = or (c == 32) (and (c >= 9) (c <= 13));

def toUpperChar c = if isLower c then c - 32 else c;
def toLowerChar c = if isUpper c then c + 32 else c;

def strToUpper s = strMap toUpperChar s;
def strToLower s = strMap toLowerChar s;

def strWords s = let go = lambda acc cur t -> match uncons t with 
        None -> if strLen cur == 0 then acc else snoc (strReverse cur) acc 
    | Just (c, rest) -> 
        if isSpace c 
        then (if strLen cur == 0 
            then go acc "" rest
            else go (snoc (strReverse cur) acc) "" rest) 
        else go acc (fromCode c ++ cur) rest in go [] "" s
    ;

def strUnwords ws = match ws with [] -> "" | w: [] -> w | w: rest -> w ++ " " ++ strUnwords rest;

def strLines s = 
    if strLen s == 0 
        then [] 
    else let go = lambda acc cur t -> match uncons t with 
          None -> if strLen cur == 0 then acc else snoc (strReverse cur) acc 
        | Just (c, rest) -> 
            if c == 10 
                then go (snoc (strReverse cur) acc) "" rest 
            else go acc (fromCode c ++ cur) rest in go [] "" s;

def strUnlines ls = match ls with [] -> "" | l: rest -> l ++ fromCode 10 ++ strUnlines rest;

def readInt s = let go = lambda acc t -> match uncons t with 
      None -> Just acc 
    | Just (c, rest) -> 
        if isDigit c 
            then go (acc * 10 + (c - 48)) rest 
        else None in match uncons s with 
              Just (45, rest) -> (
                if strLen rest == 0 
                    then None 
                else (match go 0 rest with 
                      None -> None 
                    | Just n -> Just (0 - n))
                ) 
            | _ -> go 0 s
    ;

def _ordStr a b = match (uncons a, uncons b) with
      (None, None) -> EQ
    | (None, _) -> LT
    | (_, None) -> GT
    | (Just (i, ra), Just (j, rb)) -> match intCompare i j with 
        -1 -> LT | 1 -> GT | _ -> _ordStr ra rb
    ;

implicit def ordStr = Ord { cmp = _ordStr } :: Ord String;
implicit def showStr = Show { show' = id } :: Show String;
