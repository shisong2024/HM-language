-- utils/data/map.txt

data Map k v = Leaf | Node Int k v (Map k v) (Map k v);

---

def empty = Leaf;

def height m = match m with
      Leaf -> 0
    | Node h _ _ _ _ -> h
    ;

def node k v l r = Node (1 + max (height l) (height r)) k v l r;

def lookup {o :: Ord a} k m = match m with
      Leaf -> None
    | Node _ k' v l r -> match o.cmp k k' with
          LT -> lookup k l
        | EQ -> Just v
        | GT -> lookup k r
    ;

def member k m = isJust (lookup k m);

--- 

def findMin m = match m with
      Leaf -> None
    | Node _ k v Leaf _ -> Just (k, v)
    | Node _ _ _ l _ -> findMin l
    ;

def findMax m = match m with
      Leaf -> None
    | Node _ k v _ Leaf -> Just (k, v)
    | Node _ _ _ _ r -> findMax r
    ;

def insert {o :: Ord a} k v m = match m with
      Leaf -> Node 1 k v Leaf Leaf
    | Node _ k' v' l r -> match o.cmp k k' with
          LT -> balance k' v' (insert k v l) r
        | EQ -> node k v l r
        | GT -> balance k' v' l (insert k v r)
    ;

def balance k v l r = 
    if height l > height r + 1
        then match l with
              Leaf -> Leaf
            | Node _ lk lv ll lr -> 
                if height ll >= height lr
                    then node lk lv ll (node k v lr r)
                else match lr with
                    Leaf -> Leaf
                    | Node _ lrk lrv lrl lrr -> 
                        node lrk lrv (node lk lv ll lrl) (node k v lrr r)
    else if height r > height l + 1
        then match r with
              Leaf -> Leaf
            | Node _ rk rv rl rr ->
                if height rr >= height rl
                    then node rk rv (node k v l rl) rr
                else match rl with
                      Leaf -> Leaf
                    | Node _ rlk rlv rll rlr -> 
                        node rlk rlv (node k v l rll) (node rk rv rlr rr)
    else node k v l r
    ;

def deleteMin m = match m with
      Leaf -> Leaf
    | Node _ _ _ Leaf r -> r
    | Node _ k v l r -> balance k v l (deleteMin r)
    ;

def deleteMax m = match m with
      Leaf -> Leaf
    | Node _ _ _ l Leaf -> l
    | Node _ k v l r -> balance k v (deleteMin l) r
    ;

def deleteRoot l r = match (l, r) with
      (Leaf, r) -> r
    | (l, Leaf) -> l
    | (l, r) -> match findMin r with
          None -> l
        | Just (mk, mv) -> balance mk mv l (deleteMin r)
    ;

def delete {o :: Ord a} k m = match m with
      Leaf -> Leaf
    | Node _ k' v' l r -> match o.cmp k k' with
          LT -> balance k' v' (delete k l) r
        | EQ -> deleteRoot l r
        | GT -> balance k' v' l (delete k r)
    ;

def foldr f z m = match m with
      Leaf -> z
    | Node _ k v l r -> foldr f (f k v (foldr f z r)) l
    ;

def fromList l =
    let go = lambda m l -> match l with
          [] -> m
        | (k, v): rest -> go (insert k v m) rest
    in go empty l
    ;

def map f m = match m with
      Leaf -> Leaf
    | Node h k v l r -> Node h k (f v) l r
    ;

-- union/difference/intersection
def split {o :: Ord a} k m = match m with
      Leaf -> (Leaf, None, Leaf)
    | Node _ k' v l r -> match o.cmp k k' with
          LT -> (match split k l with 
            (ll, mv, lr) -> (ll, mv, join k' v lr r))
        | EQ -> (l, Just v, r)
        | GT -> (match split k r with
            (rl, mv, rr) -> (join k' v l rl, mv, rr))
    ;

def join k v l r = match (l, r) with 
      (Leaf, _) -> insertMin k v r
    | (_, Leaf) -> insertMax k v l
    | (Node lh lk lv ll lr, Node rh rk rv rl rr) -> 
        if lh > rh + 1
            then balance lk lv ll (join k v lr r)
        else if rh > lh + 1
            then balance rk rv (join k v l rl) rr
        else node k v l r
    ;

def insertMin k v m = match m with
      Leaf -> node k v Leaf Leaf
    | Node _ k' v' l r -> balance k' v' (insertMin k v l) r
    ;

def insertMax k v m = match m with
      Leaf -> node k v Leaf Leaf
    | Node _ k' v' l r -> balance k' v' l (insertMax k v r)
    ;

def merge m1 m2 = foldr insert m1 m2;

def union m1 m2 = match m2 with
      Leaf -> m1
    | Node _ k v l r -> match split k m1 with
        (l1, mv, r1) -> 
            let l' = union l1 l in
            let r' = union r1 r in
            match mv with
                  None -> join k v l' r'
                | Just v1 -> join k v1 l' r'
    ;

def intersection m1 m2 = match m2 with
      Leaf -> Leaf
    | Node _ k v l r -> match split k m1 with
        (l1, mv, r1) ->
            let l' = intersection l1 l in
            let r' = intersection r1 r in
            match mv with
                  None -> merge l' r'
                | Just v1 -> join k v1 l' r'
    ;

def difference m1 m2 = match m2 with
      Leaf -> m1
    | Node _ k v l r -> match split k m1 with
        (l1, _, r1) -> 
            let l' = difference l1 l in
            let r' = difference r1 r in
            merge l' r'
    ;
